KraptonCard.tsx
import React, { useState, useEffect } from 'react';
import { motion } from 'framer-motion';

interface CardProps {
  title: string;
  metric: number;
}

export const KraptonCard = ({ title, metric }: CardProps) => {
  const [count, setCount] = useState(0);

  useEffect(() => {
    const timer = setInterval(() => {
      setCount(c => Math.min(c + 1, metric));
    }, 20);
    return () => clearInterval(timer);
  }, [metric]);

  return (
    <motion.div whileHover={{ scale: 1.02 }}>
      <h3>{title}</h3>
      <p>{count}%</p>
    </motion.div>
  );
};
